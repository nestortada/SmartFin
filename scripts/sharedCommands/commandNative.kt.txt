  private fun syntheticCommandFile(name: String): File {
    guard();require(name.matches(Regex("dashboard_phase2_cmd_[a-zA-Z0-9_]+\\.db")))
    return context.getDatabasePath(name)
  }
  @ReactMethod fun commandSeed(name: String, statements: ReadableArray, promise: Promise) {
    val file=syntheticCommandFile(name)
    executor.execute {try {
      file.parentFile?.mkdirs()
      val db=SQLiteDatabase.openOrCreateDatabase(file,null)
      try {db.execSQL("PRAGMA foreign_keys=ON");db.beginTransaction()
        try {for(i in 0 until statements.size())db.execSQL(statements.getString(i));db.setTransactionSuccessful()}finally{db.endTransaction()}
      }finally{db.close()}
      promise.resolve(name)
    }catch(error:Throwable){promise.reject("COMMAND_SEED",error)}}
  }
