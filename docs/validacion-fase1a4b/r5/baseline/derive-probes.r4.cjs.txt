// r4: node:sqlite writes "no such table: <t>"; the Android driver appends its result code and the statement it compiled
// ("no such table: <t> (code 1): , while compiling: <SQL>", "(code 1 SQLITE_ERROR)" on newer releases). Both are the same
// error: classified by the table name, which is kept. Only that exact shape is recognized (result code 1, the compiled
// statement must name the same table); every other message keeps its previous class or its full text.
const classBefore = `function errorClass(error: unknown): string {
  if (error === undefined) return 'accepted';
  const text = errorText(error);
`;
const classAfter = `/**
 * 1a.4B r4: name of the missing table in a "no such table" error, as written by node:sqlite ("no such table: <t>") or by
 * the Android driver ("no such table: <t> (code 1): , while compiling: <SQL naming t>"; "(code 1 SQLITE_ERROR)" or
 * "(code 1 SQLITE_ERROR[1])" on newer releases). Anything else, including a different code or a statement that does not
 * name that table, is not recognized (undefined) and keeps its full text.
 */
export function missingTableName(text: string): string | undefined {
  const match = /^no such table: ([A-Za-z_][A-Za-z0-9_]*)(?: \\(code 1(?: SQLITE_ERROR(?:\\[1\\])?)?\\): , while compiling: ([\\s\\S]+))?$/.exec(text);
  if (!match) return undefined;
  const [, table, statement] = match;
  if (statement !== undefined && !new RegExp('(^|[^A-Za-z0-9_])' + table + '([^A-Za-z0-9_]|$)').test(statement)) return undefined;
  return table;
}
export function errorClass(error: unknown): string {
  if (error === undefined) return 'accepted';
  const text = errorText(error);
  const missing = missingTableName(text);
  if (missing !== undefined) return 'other: no such table: ' + missing;
`;
